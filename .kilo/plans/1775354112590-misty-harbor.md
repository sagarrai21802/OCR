# Plan: Fix Chrome Extension for Web Store Compliance

## Problem Statement

The current extension violates Chrome Web Store policy by loading Tesseract.js dynamically from a CDN (`https://cdn.jsdelivr.net/npm/tesseract.js@5/dist/tesseract.min.js`). All code must be bundled locally.

## Current State

```
extension/
├── manifest.json     # Uses <all_urls>, no icons, loads from CDN
├── content.js     # Line 98: loads from CDN
└── popup.html
```

**Issues:**
1. Dynamic script injection from CDN (line 98 in content.js)
2. `host_permissions: "<all_urls>"` too broad
3. No icons defined
4. No `web_accessible_resources` for worker files

---

## Plan: Local Bundling Implementation

**Note:** Tesseract.js v5 uses a single combined worker, so `worker.min.js` may not be needed if bundled in the main file. Need to verify exact files required.

### Step 4: Update content.js

Replace the CDN loading function (lines 90-100) with local loading using chrome.runtime.getURL:

```javascript
// OLD (lines 90-100) - REMOVE:
function loadTesseractJS() {
    return new Promise((resolve, reject) => {
        if (window.Tesseract) {
            resolve(window.Tesseract);
            return;
        }
        
        const script = document.createElement('script');
        script.src = 'https://cdn.jsdelivr.net/npm/tesseract.js@5/dist/tesseract.min.js';
        script.onload = () => {
            console.log('[OCR AutoFill] Tesseract.js loaded from CDN');
            resolve(window.Tesseract);
        };
        script.onerror = () => reject(new Error('Failed to load Tesseract.js'));
        document.head.appendChild(script);
    });
}

// REPLACE WITH:
function getLocalTesseract() {
    if (window.Tesseract) {
        return Promise.resolve(window.Tesseract);
    }
    return Promise.reject(new Error('Tesseract.js not loaded'));
}

// Update the loadTesseract function to use local paths:
async function loadTesseract() {
    if (tesseractWorker) return tesseractWorker;
    if (tesseractLoadPromise) return tesseractLoadPromise;
    
    console.log('[OCR AutoFill] Initializing Tesseract...');
    
    tesseractLoadPromise = (async () => {
        // Wait for Tesseract to be available (loaded via script tag in manifest)
        await new Promise(resolve => {
            if (window.Tesseract) resolve();
            else {
                const checkTesseract = setInterval(() => {
                    if (window.Tesseract) {
                        clearInterval(checkTesseract);
                        resolve();
                    }
                }, 100);
            }
        });
        
        const basePath = chrome.runtime.getURL('libs');
        
        tesseractWorker = await Tesseract.createWorker('eng', 1, {
            workerPath: basePath + '/worker.min.js',
            langPath: basePath + '/',
            corePath: basePath + '/tesseract-core.wasm.js',
            logger: m => {
                if (m.status === 'recognizing text') {
                    console.log(`[OCR AutoFill] OCR Progress: ${Math.round(m.progress * 100)}%`);
                }
            }
        });
        
        console.log('[OCR AutoFill] Tesseract worker ready');
        return tesseractWorker;
    })();
    
    return tesseractLoadPromise;
}
```
